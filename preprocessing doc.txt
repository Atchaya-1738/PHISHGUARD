def extract_body(msg):
    """
    Recursively walk a parsed email message and extract plain-text content.
    Falls back to stripping HTML tags if only HTML parts exist.
    """
    body_parts = []

    if msg.is_multipart():
        for part in msg.walk():
            ctype = part.get_content_type()
            disposition = str(part.get('Content-Disposition', ''))
            if 'attachment' in disposition:
                continue   # skip file attachments
            if ctype == 'text/plain':
                payload = part.get_payload(decode=True)
                if payload:
                    charset = part.get_content_charset() or 'utf-8'
                    body_parts.append(payload.decode(charset, errors='replace'))
            elif ctype == 'text/html':
                payload = part.get_payload(decode=True)
                if payload:
                    charset = part.get_content_charset() or 'utf-8'
                    soup = BeautifulSoup(payload.decode(charset, errors='replace'), 'html.parser')
                    body_parts.append(soup.get_text(separator=' '))
    else:
        # Single-part email
        payload = msg.get_payload(decode=True)
        if payload:
            charset = msg.get_content_charset() or 'utf-8'
            text = payload.decode(charset, errors='replace')
            if msg.get_content_type() == 'text/html':
                soup = BeautifulSoup(text, 'html.parser')
                text = soup.get_text(separator=' ')
            body_parts.append(text)

    return ' '.join(body_parts).strip()


def parse_email_file(filepath):
    """
    Read a raw email file and return a dict of useful fields.
    """
    try:
        with open(filepath, 'rb') as f:
            msg = BytesParser(policy=policy.compat32).parse(f)

        subject   = msg.get('Subject', '') or ''
        sender    = msg.get('From', '')    or ''
        recipient = msg.get('To', '')      or ''
        body      = extract_body(msg)

        # Combined text field (subject + body) for NLP
        full_text = subject + ' ' + body

        return {
            'subject':   subject,
            'sender':    sender,
            'recipient': recipient,
            'body':      body,
            'full_text': full_text,
        }
    except Exception as e:
        # Return empty strings on parse failure; we'll drop these later
        return {'subject': '', 'sender': '', 'recipient': '', 'body': '', 'full_text': ''}

def clean_text(text):
    """
    Full text-cleaning pipeline for one email string.
    Returns a cleaned lowercase string.
    """
    if not isinstance(text, str):
        return ''

    # 1. Lowercase
    text = text.lower()

    # 2. Remove URLs  (http://... or www...)
    text = re.sub(r'http\S+|www\.\S+', ' url ', text)

    # 3. Remove email addresses
    text = re.sub(r'\S+@\S+', ' email ', text)

    # 4. Strip any leftover HTML tags
    text = re.sub(r'<[^>]+>', ' ', text)

    # 5. Remove numbers (optional — change to keep if you want numeric features)
    text = re.sub(r'\d+', ' ', text)

    # 6. Keep only alphabetic characters and spaces
    text = re.sub(r'[^a-z\s]', ' ', text)

    # 7. Collapse multiple spaces into one
    text = re.sub(r'\s+', ' ', text).strip()

    return text

STOP_WORDS = set(stopwords.words('english'))

def tokenize_and_remove_stopwords(text):
    """
    Tokenize text and remove English stopwords.
    Returns a list of tokens.
    """
    tokens = word_tokenize(text)           # NLTK punkt tokenizer
    tokens = [t for t in tokens
               if t not in STOP_WORDS      # remove stopwords
               and len(t) > 1]             # remove single characters
    return tokens

def extract_structural_features(row):
    """
    Compute handcrafted numeric features from raw email fields.
    """
    body    = str(row.get('body', ''))
    subject = str(row.get('subject', ''))
    sender  = str(row.get('sender', ''))

    return {
        # Length features
        'body_length':         len(body),
        'subject_length':      len(subject),
        'num_words_body':      len(body.split()),

        # Spam signal features
        'num_exclamations':    body.count('!') + subject.count('!'),
        'num_question_marks':  body.count('?'),
        'num_dollar_signs':    body.count('$'),
        'has_html':            int(bool(re.search(r'<[a-z][\s\S]*?>', body, re.I))),
        'num_urls':            len(re.findall(r'http\S+|www\.\S+', body, re.I)),
        'num_email_addresses': len(re.findall(r'\S+@\S+', body)),
        'subject_all_caps':    int(bool(re.search(r'\b[A-Z]{3,}\b', subject))),
        'subject_has_digits':  int(bool(re.search(r'\d', subject))),
        'subject_has_free':    int('free' in subject.lower()),
        'sender_is_numeric':   int(bool(re.search(r'\d{4,}', sender))),
    }
